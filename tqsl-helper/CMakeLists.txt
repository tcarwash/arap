cmake_minimum_required(VERSION 3.14)
project(arap_tqsl_sign LANGUAGES C CXX)
include(GNUInstallDirs)
option(ARAP_BUILD_FIXTURE_IMPORTER "Build the isolated synthetic-certificate fixture importer" OFF)
if(NOT APPLE AND NOT WIN32)
  if(ARAP_BUILD_FIXTURE_IMPORTER)
    add_compile_definitions(CONFDIR="${TQSL_SOURCE_DIR}/src/")
  else()
    add_compile_definitions(CONFDIR="${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_DATADIR}/TrustedQSL/")
  endif()
endif()

if(NOT TQSL_SOURCE_DIR OR NOT EXISTS "${TQSL_SOURCE_DIR}/src/tqsllib.h")
  message(FATAL_ERROR "Set TQSL_SOURCE_DIR to a TrustedQSL source checkout")
endif()

find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
file(STRINGS "${TQSL_SOURCE_DIR}/apps/tqslversion.ver" TQSL_SOURCE_VERSION)
if(NOT TQSL_SOURCE_VERSION MATCHES "^([0-9]+)\\.([0-9]+)(\\.([0-9]+))?$")
  message(FATAL_ERROR "Unable to parse TrustedQSL apps/tqslversion.ver")
endif()
set(TQSL_VERSION_MAJOR "${CMAKE_MATCH_1}")
set(TQSL_VERSION_MINOR "${CMAKE_MATCH_2}")
set(TQSL_VERSION_UPDATE "${CMAKE_MATCH_4}")
if(NOT TQSL_VERSION_UPDATE)
  set(TQSL_VERSION_UPDATE 0)
endif()
set(TQSL_VERSION_PATCH 0)
set(TQSLLIB_VERSION_MAJOR ${TQSL_VERSION_MAJOR})
set(TQSLLIB_VERSION_MINOR ${TQSL_VERSION_MINOR})
set(TQSLLIB_VERSION_UPDATE ${TQSL_VERSION_UPDATE})
set(TQSLLIBNAME tqsllib)
set(BUILD_GENCRQ OFF)
set(BUILD_LOADCERT OFF)
set(BUILD_STATIONLOC OFF)
set(BUILD_CONVERTER OFF)

add_subdirectory("${TQSL_SOURCE_DIR}/src" tqsllib)
add_executable(arap-tqsl-sign main.cpp)
target_include_directories(arap-tqsl-sign PRIVATE "${TQSL_SOURCE_DIR}/src")
target_link_libraries(arap-tqsl-sign PRIVATE tqsllib OpenSSL::Crypto)
if(APPLE)
  set_target_properties(arap-tqsl-sign PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(arap-tqsl-sign PROPERTIES BUILD_RPATH "$ORIGIN/tqsllib;$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()

if(ARAP_BUILD_FIXTURE_IMPORTER)
  add_executable(arap-tqsl-fixture-import fixture_import.cpp)
  target_include_directories(arap-tqsl-fixture-import PRIVATE "${TQSL_SOURCE_DIR}/src")
  target_link_libraries(arap-tqsl-fixture-import PRIVATE tqsllib OpenSSL::Crypto)
  set_target_properties(arap-tqsl-fixture-import PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:tqsllib>")
endif()
